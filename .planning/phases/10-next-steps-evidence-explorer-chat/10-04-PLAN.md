**Step A — Add to `src/frontend/data/policies.ts`:**

Add interface and function without modifying existing exports:

```ts
export interface EvidenceSearchResult {
  policyId: string;
  policyTitle: string;
  payer: string;
  drugFamily: string;
  version: number;
  fieldLabel: string;
  snippet: string;
  document: string;
  page: number | null;
  section: string;
}

export interface EvidenceSearchResponse {
  query: string;
  count: number;
  results: EvidenceSearchResult[];
}

export async function fetchEvidenceSearch(query: string): Promise<EvidenceSearchResponse> {
  const res = await fetch(`/api/evidence/search?q=${encodeURIComponent(query)}`);
  if (!res.ok) throw new Error(`Search failed: ${res.status}`);
  return res.json() as Promise<EvidenceSearchResponse>;
}
```
