**Step A — Create `src/server/policy-qa-chat.ts`:**

This module wraps the Phase 3 hybrid QA evidence retrieval and LLM answering into a plain Express route handler.

Study the existing `src/mcp/tools/ask_policy_question.ts` to understand:
- `retrieveEvidence(question, keywords)` from `./utils/evidence_retriever.js`
- `answerWithGrounding(question, evidence)` from `./utils/llm_client.js`
- `extractQuestionKeywords(question)` from `./utils/evidence_retriever.js`
- The `EvidenceItem` type from `../utils/response_builder.js`

Create:

```ts
import type { Express, Request, Response } from 'express';
import { extractQuestionKeywords, retrieveEvidence } from '../mcp/tools/utils/evidence_retriever.js';
import { answerWithGrounding } from '../mcp/tools/utils/llm_client.js';
import type { EvidenceItem } from '../mcp/utils/response_builder.js';

export interface ChatEvidenceCitation {
  policyId: string;
  policyName: string;
  page: number | null;
  section: string;
  snippet: string;
}

export interface PolicyQaChatResponse {
  answer: string;                       // Full answer text with inline citations
  evidence: ChatEvidenceCitation[];     // Evidence items cited
  query: string;
}

export function registerPolicyQaChatRoute(app: Express): void {
  app.post('/api/chat/policy-qa', async (req: Request, res: Response) => {
    const { question } = req.body as { question?: string };
    if (!question || !question.trim()) {
      res.status(400).json({ error: 'question is required' });
      return;
    }

    try {
      const keywords = extractQuestionKeywords(question);
      const evidenceItems: EvidenceItem[] = retrieveEvidence(question, keywords);

      // Build answer using the existing grounding function
      const rawAnswer = await answerWithGrounding(question, evidenceItems);

      // Convert EvidenceItem[] to ChatEvidenceCitation[]
      // EvidenceItem shape from response_builder.ts — read the type to map fields correctly
      const citations: ChatEvidenceCitation[] = evidenceItems.map((item) => ({
        policyId: item.policyId ?? '',
        policyName: item.policyName ?? item.policyId ?? '',
        page: typeof item.page === 'number' ? item.page : null,
        section: item.section ?? '',
        snippet: item.snippet ?? item.text ?? '',
      }));

      // Append inline citation markers to the answer text for each unique policy referenced
      // Format: "(PolicyName · page N · Section)" appended at the end of the answer paragraph
      const uniquePolicies = [...new Map(citations.map((c) => [c.policyId, c])).values()];
      const citationSuffix = uniquePolicies
        .filter((c) => c.policyName)
        .map((c) => `(${c.policyName}${c.page ? ` · page ${c.page}` : ''}${c.section ? ` · ${c.section}` : ''})`)
        .join(' ');
      const answerWithCitations = citationSuffix ? `${rawAnswer}\n\n${citationSuffix}` : rawAnswer;

      const payload: PolicyQaChatResponse = {
        answer: answerWithCitations,
        evidence: citations,
        query: question.trim(),
      };

      res.json(payload);
    } catch (err) {
      const message = err instanceof Error ? err.message : 'Unknown error';
      res.status(500).json({ error: message });
    }
  });
}
```

Note: Read `src/mcp/utils/response_builder.ts` to see the exact `EvidenceItem` interface shape before writing the field mapping — adapt `item.policyId`, `item.policyName`, `item.page`, `item.section`, `item.snippet` or `item.text` to match whatever fields actually exist on the type.

**Step A — Create `src/frontend/components/chat-view.tsx`:**

Types:
```ts
type ChatMessage = {
  id: string;
  role: 'user' | 'assistant';
  content: string;
  evidence?: ChatEvidenceCitation[];
};
type ChatEvidenceCitation = {
  policyId: string;
  policyName: string;
  page: number | null;
  section: string;
  snippet: string;
};
```

State:
- `messages: ChatMessage[]` — session-only history (starts empty, not persisted)
- `input: string` — controlled textarea value
- `isLoading: boolean`
- `error: string | null`
- `activeEvidence: { title: string; evidence: PolicyEvidenceRef[] } | null` — for the sidebar panel

Key behavior:
- On submit: append a `{ id, role: 'user', content: input }` message to `messages`, clear input, set loading, call `POST /api/chat/policy-qa` with `{ question: input }`, on success append `{ role: 'assistant', content: response.answer, evidence: response.evidence }` and set `activeEvidence` to the response evidence mapped to `PolicyEvidenceRef[]`
- Map `ChatEvidenceCitation` to `PolicyEvidenceRef` for the existing `PolicyEvidencePanel`: `{ id: idx.toString(), snippet: c.snippet, document: c.policyName, page: c.page ?? undefined, section: c.section }`
- On error: set error message, do NOT add assistant message

Render layout: two-column flex when evidence is open, single column otherwise:
```tsx
<div className="chat-layout">
  <div className="chat-main">
    <h1>Chat</h1>
    <p className="eyebrow">Ask natural-language questions about loaded policies</p>
    <div className="chat-messages" ref={messagesEndRef}>
      {messages.map((msg) => (
        <div key={msg.id} className={`chat-message chat-message-${msg.role}`}>
          <p>{msg.content}</p>
          {msg.role === 'assistant' && msg.evidence && msg.evidence.length > 0 && (
            <button
              type="button"
              className="btn-link"
              onClick={() => setActiveEvidence({
                title: 'Sources',
                evidence: msg.evidence!.map((c, i) => ({ id: String(i), snippet: c.snippet, document: c.policyName, page: c.page ?? undefined, section: c.section }))
              })}
            >
              View {msg.evidence.length} source{msg.evidence.length !== 1 ? 's' : ''}
            </button>
          )}
        </div>
      ))}
      {isLoading && <div className="chat-message chat-message-assistant"><p>Thinking…</p></div>}
      {error && <p className="error-text">{error}</p>}
    </div>
    <form className="chat-input-form" onSubmit={handleSubmit}>
      <textarea
        className="chat-input"
        value={input}
        onChange={(e) => setInput(e.target.value)}
        onKeyDown={(e) => { if (e.key === 'Enter' && !e.shiftKey) { e.preventDefault(); void handleSubmit(e); } }}
        placeholder="Ask about prior authorization, step therapy, coverage criteria…"
        rows={3}
        disabled={isLoading}
      />
      <button type="submit" className="btn-primary" disabled={isLoading || !input.trim()}>Send</button>
    </form>
  </div>
  {activeEvidence && (
    <PolicyEvidencePanel
      title={activeEvidence.title}
      evidence={activeEvidence.evidence}
      onClose={() => setActiveEvidence(null)}
    />
  )}
</div>
```
