export async function fetchNextSteps(evalId: string): Promise<NextStepsPayload> {
  const res = await fetch(`/api/patients/evaluations/${evalId}/next-steps`);
  if (!res.ok) throw new Error(`Failed to load next steps: ${res.status}`);
  return res.json() as Promise<NextStepsPayload>;
}
