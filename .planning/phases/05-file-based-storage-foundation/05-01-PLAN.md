  <name>Task 1: Create storage types module</name>
  <files>src/storage/types.ts</files>
  <action>
    Create `src/storage/types.ts` exporting all v2.0 storage entity types. Do NOT redefine PolicyRecord — import it from `../../data/schemas/policy.schema.js` and reference it.

    Export these interfaces/types:

    ```ts
    // Re-export for convenience
    export type { PolicyRecord } from '../../data/schemas/policy.schema.js';

    // One versioned file = one PolicyVersion
    export interface PolicyVersion {
      policyId: string;          // e.g. "bcbs-nc-bevacizumab"
      version: number;           // 1-based integer
      fileName: string;          // e.g. "bcbs-nc-bevacizumab_v1.json"
      savedAt: string;           // ISO 8601 timestamp
      record: PolicyRecord;      // Full normalized policy
    }

    // Entry in index.json per policy
    export interface PolicyIndexEntry {
      policyId: string;
      payer: string;
      title: string;
      drugFamily: string;        // drug.brandName from PolicyRecord
      versions: number[];        // e.g. [1, 2, 3]
      currentVersion: number;    // points to latest
      currentVersionFile: string; // e.g. "bcbs-nc-bevacizumab_v3.json"
      lastUpdatedAt: string;
    }

    // Top-level index.json structure
    export interface PolicyIndex {
      updatedAt: string;
      policies: PolicyIndexEntry[];
    }

    // Field-level diff between two policy versions
    export interface DiffField {
      field: string;         // dot-path e.g. "drug.brandName"
      oldValue: unknown;
      newValue: unknown;
    }

    // Stored alongside versioned policy files
    export interface DiffRecord {
      policyId: string;
      fromVersion: number;
      toVersion: number;
      timestamp: string;
      changes: DiffField[];
    }

    // Patient case folder metadata file: data/patients/{caseId}/case.json
    export interface PatientCase {
      caseId: string;            // UUID
      payer: string;
      requestedDrug: string;
      diagnosis: string;
      patientName: string;       // synthetic/de-identified only
      status: 'missing-docs' | 'ready-for-eval' | 'complete';
      createdAt: string;
      updatedAt: string;
      documentFiles: string[];   // filenames inside case folder
      extractedFactsFile?: string; // e.g. "facts.json" — present after extraction
    }

    // Stored as data/evaluations/{evalId}.json
    export interface CoverageEvaluation {
      evalId: string;            // UUID
      caseId: string;
      policyId: string;
      policyVersion: number;
      evaluatedAt: string;
      coverageStatus: 'Covered' | 'PA Required' | 'Likely Eligible but Docs Missing' | 'Not Covered' | 'Preferred Alternative Required' | 'Unclear';
      checklist: EvaluationChecklistItem[];
    }

    export interface EvaluationChecklistItem {
      criterion: string;
      status: 'PASS' | 'MISSING' | 'UNKNOWN' | 'NEEDS REVIEW';
      matchedFact?: string;
      evidenceSnippet?: string;
    }
    ```

  <action>
    Create `src/storage/paths.ts` with centralised path constants derived from the project root using `import.meta.url` (ESM pattern — same as existing `src/server/ingestion/store.ts`).

    ```ts
    import { mkdirSync } from 'fs';
    import { dirname, join } from 'path';
    import { fileURLToPath } from 'url';

    const __filename = fileURLToPath(import.meta.url);
    const __dirname = dirname(__filename);

    // Project root is two levels up from src/storage/
    export const PROJECT_ROOT = join(__dirname, '../..');

    export const DATA_ROOT              = join(PROJECT_ROOT, 'data');
    export const POLICIES_RAW_DIR       = join(DATA_ROOT, 'policies', 'raw');
    export const POLICIES_STRUCTURED_DIR = join(DATA_ROOT, 'policies', 'structured');
    export const POLICY_INDEX_PATH      = join(DATA_ROOT, 'policies', 'index.json');
    export const PATIENTS_DIR           = join(DATA_ROOT, 'patients');
    export const EVALUATIONS_DIR        = join(DATA_ROOT, 'evaluations');

    /**
     * Ensure all v2.0 data directories exist.
     * Safe to call multiple times (mkdirSync with recursive:true is idempotent).
     */
    export function ensureDataDirectories(): void {
      mkdirSync(POLICIES_RAW_DIR,        { recursive: true });
      mkdirSync(POLICIES_STRUCTURED_DIR, { recursive: true });
      mkdirSync(PATIENTS_DIR,            { recursive: true });
      mkdirSync(EVALUATIONS_DIR,         { recursive: true });
    }
    ```

    After creating the file, verify directories are created by importing and calling `ensureDataDirectories()` in a one-shot test:
    ```bash
    node --input-type=module <<'EOF'

    ensureDataDirectories();
    console.log('OK');
    EOF
    ```
    (Run from project root with tsx if needed: `npx tsx -e "import('./src/storage/paths.ts').then(m => { m.ensureDataDirectories(); console.log('OK'); })"`)
