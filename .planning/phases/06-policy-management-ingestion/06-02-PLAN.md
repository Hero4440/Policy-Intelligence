      lastUpdatedAt: string;

    Exported functions:

    **fetchPolicies(): Promise&lt;PolicyIndexEntry[]&gt;**
    - GET /api/policies
    - Returns `data.policies` array, or `[]` on error

    **fetchPolicyDetail(policyId: string, version?: number): Promise&lt;{ entry: PolicyIndexEntry; record: unknown } | null&gt;**
    - GET /api/policies/:policyId?version=N (omit version param if not provided)
    - Returns `{ entry, record }` or null on 404/error

    **fetchPolicyVersions(policyId: string): Promise&lt;{ versions: number[]; currentVersion: number }&gt;**
    - GET /api/policies/:policyId/versions
    - Returns `{ versions: [], currentVersion: 0 }` on error

    **uploadPolicyPdf(fileName: string, fileBuffer: ArrayBuffer): Promise&lt;{ policyId: string; version: number; record: unknown }&gt;**
    - Converts buffer to base64 string: `btoa(String.fromCharCode(...new Uint8Array(fileBuffer)))`
    - POST /api/policies/upload with body `{ fileName, contentBase64 }`
    - Returns `{ policyId, version, record }` or throws on error

    All functions use fetch with `Content-Type: application/json`. All catch blocks return safe defaults (log error via console.error). No environment variable needed — paths are relative (`/api/...`).

    Computed:
    ```typescript
    const uniquePayers = [...new Set(entries.map(e => e.payer))].sort();
    const uniqueDrugFamilies = [...new Set(entries.map(e => e.drugFamily))].sort();
    const filtered = entries
      .filter(e => !filterPayer || e.payer === filterPayer)
      .filter(e => !filterDrugFamily || e.drugFamily === filterDrugFamily);
    ```

    Render structure (use existing CSS classes from styles.css — `panel-header`, `detail-card`, `empty-state`, `eyebrow`):
    ```
    <div className="policy-rules-page">
      <div className="panel-header">
        <div>
          <p className="eyebrow">Policy Management</p>
          <h2>Policy Rules</h2>
        </div>
        <span className="panel-count">{filtered.length} policies</span>
      </div>

      <div className="policy-filters">
        <select value={filterPayer} onChange={...}>
          <option value="">All payers</option>
          {uniquePayers.map(p => <option key={p} value={p}>{p}</option>)}
        </select>
        <select value={filterDrugFamily} onChange={...}>
          <option value="">All drug families</option>
          {uniqueDrugFamilies.map(d => <option key={d} value={d}>{d}</option>)}
        </select>
      </div>

      {isLoading && <div className="empty-state">Loading policies...</div>}
      {error && <div className="chat-error">{error}</div>}
      {!isLoading && filtered.length === 0 && (
        <div className="empty-state">No policies loaded yet. Upload a PDF to get started.</div>
      )}
      {filtered.map(entry => (
        <article key={entry.policyId} className="detail-card" style={{ cursor: 'pointer' }}
          onClick={() => onSelectPolicy(entry.policyId)}>
          <p className="detail-card-title">{entry.title}</p>
          <p>{entry.payer} · {entry.drugFamily}</p>
          <small>v{entry.currentVersion} · Updated {new Date(entry.lastUpdatedAt).toLocaleDateString()}</small>
        </article>
      ))}
    </div>
    ```
